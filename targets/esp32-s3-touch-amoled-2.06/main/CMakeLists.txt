# Write a file only when its content would change. Preserves mtime when content matches,
# so downstream make/ninja rules don't see spurious changes on every reconfigure.
function(_gea_write_if_different FILE_PATH CONTENT)
    set(_should_write 1)
    if(EXISTS "${FILE_PATH}")
        file(READ "${FILE_PATH}" _existing)
        if("${_existing}" STREQUAL "${CONTENT}")
            set(_should_write 0)
        endif()
    endif()
    if(_should_write)
        file(WRITE "${FILE_PATH}" "${CONTENT}")
    endif()
endfunction()

# and the app code that references them now lives in the separate libgea_active_geatsc.a

# satisfy a cross-archive reference, so the assets would link out and leave undefined
# references (gea_asset_*_png). Instead it is built into its own static lib that is linked
# with --whole-archive below, forcing the arrays in regardless of the weak attribute
# (which is still needed to deduplicate shared assets).

set(GEATSC_COMPILE_OPTIONS
    -Wno-unused-variable
    -Wno-return-type
    -Wno-format-truncation
    # Generated destructuring emits an empty shared_ptr fallback branch that is
    # dead for required fields. Only this board compiles geatsc at -O2, and the
    # extra inlining lets gcc 15 trace that branch into a member push_back and
    # report a bogus "subscript 0 is outside array bounds" on the null object.
    -Wno-array-bounds
    # Chained property readers lower to nested decltype(auto) helpers where an
    # inner reader's missing-value branch returns a fresh temporary while its
    # present-value branch returns a reference into the (persistent) store data.
    # gcc 13+ -Wdangling-reference then flags the outer `auto&&` binding as
    # possibly dangling. It only dangles on the missing-value path, which the
    # weather store never takes (the fetched payload always carries the field),
    # and the warning is a well-known false-positive generator for exactly this
    # decltype(auto)-chain shape. Suppress here like -Wno-array-bounds above.
    -Wno-dangling-reference
    # Same reader chain, GCC's other dangling diagnostic: inlining .size() on the
    # `auto&&` binding of a reader whose missing-value branch returns a temporary
    # std::vector makes gcc 15 read it as a dangling pointer into that temporary's
    # storage. Same never-taken branch, same false positive.
    -Wno-dangling-pointer
    # And its third face: gcc 15 computes .size() (_M_finish - _M_start) on that
    # same missing-branch temporary vector and calls the members uninitialized.
    # Scoped to the generated GEATSC archive only (this block never applies to
    # framework or app source), so hand-written uninitialized bugs still error.
    -Wno-uninitialized
    -O2
    -mtext-section-literals
)

# Note: BLE driver registration + BLE controller preinit are forced from
# `app_main.cpp` and `bluetooth_service.cpp` regardless of the macro above,
# because `geatsc analyze` misses BLE usage routed through the internal
# `runtime-surface` re-export (as `gea-embedded`'s Settings panel does).
# The macro is kept narrow because `app_runner.cpp` uses it to decide
# whether `gea_init` is allocated in SPIRAM (when undefined) or internal
# DRAM (when defined). With 64 KB gea_init stack, forcing internal DRAM
# fails — only 32 KB largest contiguous block remains after wifi/diag init.
# `_GEA_EMBEDDED_HOST_FEATURES` is populated by `geatsc analyze` for the
# application.
#
# The user-facing `GEA_EMBEDDED_HTTPS_SUPPORT` cache option lets the build
# override auto-detection: `on` forces HTTPS linkage even when no
# `https://` URL appears in source, `off` skips it even when one does.
# `auto` (default) trusts analyze's verdict.
string(TOLOWER "${GEA_EMBEDDED_HTTPS_SUPPORT}" _GEA_EMBEDDED_HTTPS_SUPPORT_MODE)
set(_GEA_EMBEDDED_ENABLE_HTTPS 0)
if(_GEA_EMBEDDED_HTTPS_SUPPORT_MODE STREQUAL "auto")
    if("${_GEA_EMBEDDED_HOST_FEATURES}" MATCHES "(^|;)https($|;)")
        set(_GEA_EMBEDDED_ENABLE_HTTPS 1)
    endif()
elseif(_GEA_EMBEDDED_HTTPS_SUPPORT_MODE STREQUAL "on"
        OR _GEA_EMBEDDED_HTTPS_SUPPORT_MODE STREQUAL "true"
        OR _GEA_EMBEDDED_HTTPS_SUPPORT_MODE STREQUAL "yes"
        OR _GEA_EMBEDDED_HTTPS_SUPPORT_MODE STREQUAL "1")
    set(_GEA_EMBEDDED_ENABLE_HTTPS 1)
elseif(_GEA_EMBEDDED_HTTPS_SUPPORT_MODE STREQUAL "off"
        OR _GEA_EMBEDDED_HTTPS_SUPPORT_MODE STREQUAL "false"
        OR _GEA_EMBEDDED_HTTPS_SUPPORT_MODE STREQUAL "no"
        OR _GEA_EMBEDDED_HTTPS_SUPPORT_MODE STREQUAL "0")
    set(_GEA_EMBEDDED_ENABLE_HTTPS 0)
else()
    message(FATAL_ERROR "GEA_EMBEDDED_HTTPS_SUPPORT must be auto, on, or off")
endif()

if(_GEA_EMBEDDED_ENABLE_HTTPS AND GEA_EMBEDDED_CAPABILITY_NETWORK)

    # `mbedtls` lands in REQUIRES iff the active app actually needs HTTPS.
    # The historical reason for unconditionally listing it (script-mode
    # discrepancy with `GEA_EMBEDDED_APP=tic-tac-toe`) is gone now that
    # script-mode falls through to a feature-rich gea placeholder. The
    # condition here is the actual signal: an app with no https:// in source

# CS-held (non-draining) framebuffer stream: pipeline the fused-raster chunks via
# queueColorStreamData (raster chunk N+1 overlaps chunk N's DMA) exactly like the
# present path that gets non-jsx to 65fps — instead of the synchronous per-chunk
# txColor that drains the SPI queue every chunk. MEASURED 2026-07-08 net-neutral:
# CS-held at 36-row/14-chunk = 58.3fps vs draining 58.7 (noise). The fused flush is
# NOT drain-bound — pipelining doesn't help. The ~1.6ms gap vs non-jsx's present
# path (11.9ms/65fps) is the RASTER ROUTINE: jsx replays a full reactive display
# list (64 nodes: clip/style/layout) per chunk vs present's lean command list.
# Closing it needs a fused-raster→command-list rewrite, not drain-vs-CS-held.
# Left 0 (draining path proven); code kept in display.cpp gated off. Do not retry
# CS-held at other chunk sizes: 16-row = 47.5fps (per-chunk overhead x31 dominates).
# PROBE RESULT 2026-07-08: rasterSum~4350us / flushWall~11800us — fused raster is only
# 37% of the flush and FULLY HIDDEN under DMA slot-waits. Flush is WIRE-BOUND at ~34MB/s
# (=present's rate). The ~1.7ms/frame beginColorStream bus-acquire cancels CS-held's
# pipelining gain → net-neutral. Conclusion: a fused-raster→command-list rewrite buys

    # multiply the per-chunk fused-raster cost. The bounding-box / adaptive-coalesce
    # flush is the correct design here; the 51->60 gap is retained COMPUTE, not flush.
    # (kept DIRTY_REGION_MAX_RECTS at its 32 default; raising it only helped precision,
    # which the flush then undoes for good reason.)
    # This board's flush cost is per-pixel (PSRAM->staging copy) with ~0 per-window setup,
    # so coalescing scattered dirty rects into gap-filled bands is pure waste — it spiked the
    # copy to ~10ms on wide-dirty (spread-balls) frames and missed the TE. Tighten the dirty
    # gap-merge to near-touching rects only so the flush copies the balls, not the gaps.
    GEA_EMBEDDED_DIRTY_COALESCE_GAP_SHIFT=3
    # Old-spine comparison mode: persistent framebuffer path, no replay-fused
    # flush. This keeps canvas-3d on the path used by the 70-90fps historical
    # measurements while we compare the newer runtime/binary size separately.
    GEA_EMBEDDED_DISPLAY_FUSE_REPLAY_FLUSH=${GEA_EMBEDDED_FUSE_FLUSH_VALUE}
    # Golden (e6614f6a, 62fps) used no partial-flush edge pad and showed no smudge.
    # The 2px all-sides pad inflates every per-rect ball flush window (~24/frame:
    # 15k -> 24k px). Drop it; if the CO5300 left-edge smudge reappears it's a
    # separate regression to fix at its source, not by padding every flush.
    GEA_EMBEDDED_DISPLAY_PARTIAL_FLUSH_EDGE_PAD_PX=0
    # QSPI display SPI clock. 80 MHz is the panel's proven rate and the source
    # default (display.cpp). Pinned explicitly here because the build had drifted
    # to an effective 60 MHz (no CMake/cache/source basis — a stale configure
    # artifact). A full-screen flush is 410x502x2B = 411 KB, bandwidth-bound over
    # the 4 QSPI lanes, so the clock sets flush time linearly: 80 MHz (~40 MB/s,
    # ~10.3 ms) vs 60 MHz (~30 MB/s, ~13.7 ms). Dominates full-screen canvas apps.
    GEA_EMBEDDED_DISPLAY_QSPI_PCLK_HZ=80000000
    # Direct PSRAM->SPI framebuffer flush: DISABLED — we use the
    # PSRAM->internal-staging->SPI path instead. Exhaustively tested (PSRAM at
    # 80/60/40/20 MHz, GDMA burst 32/64B, and PSRAM overclocked to 120 MHz): the
    # GDMA can't keep the 64-byte SPI TX FIFO fed straight from octal PSRAM for a
    # full-frame stream, so it underflows, and the underflow recovery hangs the
    # frame task. Staging is the correct, stable 60 fps path on this board.
    # STATIC_BACKDROP: ON. Worth ~18 fps here; two wrong turns are recorded so
    # neither is repeated.
    #
    # Wrong turn 1 -- blamed for a rendering artifact. On 2026-09-15 this flag was
    # switched off to chase byte-swapped magenta/orange pixels along css-3d-cube's
    # face seams, on the theory that the per-frame blit window stranded stale colours
    # at a CO5300 dropped column. The flag was innocent. Those pixels came from
    # scalarPx in drawTransformedLinearGradient pairing a toRgb565 read (panel ->
    # normal) with a toNative write. toNative is the FORMAT converter and is the
    # identity on RGB565, so it applies NO panel-endian swap -- every pixel that path
    # wrote landed byte-swapped. It showed only at span heads/tails (the cube
    # silhouette) and on geometry too thin to reach the aligned PIE path (the 2px
    # horizon), which is what made it look like a seam artifact. Fixed in core
    # (render.cpp); with that fix the corruption is gone AND the backdrop is on.
    #
    # Wrong turn 2 -- reading the absence of [bdrop] lines as "never engages".
    # It means the opposite. The cache bakes ONCE at settle and a kQuiet frame KEEPS
    # it (tree_render.cpp), so a healthy steady state logs nothing at all; a log line
    # appears only on a bake or a drop. Nor does recprobe fullrebuilds:90/90 mean the
    # path is idle -- that counts display-list RECORDS, which is a different question
    # from whether the backdrop pixels are blitted. The counter that does answer it is
    # `tree(count:N direct:N)`, and it is a PER-FRAME average (treeRefreshCalls /
    # frameCount, treeDirectReplayCalls / frameCount -- frame_scheduler.cpp), so
    # direct:1 is "direct replay on every refreshing frame", not "1 frame in 90".
    #
    # Measured, css-3d-cube translucent, 55s windows: ON 50-56 fps (mean 52),
    # OFF 31-44 fps (mean 34). Turning it off also switches off the transform
    # reproject, which is gated on staticBackdropActive().
    GEA_EMBEDDED_DISPLAY_STATIC_BACKDROP=1
    GEA_EMBEDDED_DISPLAY_DIRECT_PSRAM_FLUSH=0
    # No mid-stream pacing on full-frame flushes. PACE_CHUNKS>0 forces a full
    # waitForFlushComplete() drain every N RAMWRC chunks. Measured: dropping
    # from 4 to 0 only saved ~0.8 ms of the ~6 ms per-flush wait — most of that
    # wait is 2-slot flush-semaphore backpressure (16-row pool, internal-RAM
    # starved), not the pace drain. Still keep 0: the semaphore already
    # backpressures the queue so the pace is redundant, and it's the source
    # default. (Was =4 to reproduce an old canvas-3d spine baseline.)
    GEA_EMBEDDED_DISPLAY_CO5300_STREAM_PACE_CHUNKS=0
    GEA_EMBEDDED_DISPLAY_DIRECT_PSRAM_STREAM_PACE_CHUNKS=0
    GEA_EMBEDDED_DISPLAY_DIRECT_PSRAM_RAMWRC_CONTINUE=1
    GEA_EMBEDDED_DISPLAY_DIRECT_PSRAM_CS_HELD_STREAM=1
    GEA_EMBEDDED_DISPLAY_DIRECT_PSRAM_MAX_TRANSFER_BYTES=32256
    GEA_EMBEDDED_DISPLAY_DIRECT_PSRAM_STATIC_TX_PROBE=0
    GEA_EMBEDDED_DISPLAY_DIRECT_PSRAM_STATIC_TX_PROBE_IDLE_FLUSH=0
    GEA_EMBEDDED_DISPLAY_DIRECT_PSRAM_EDGE_GUARD_BYTES=0
    GEA_EMBEDDED_DISPLAY_DIRECT_PSRAM_STAGED_PREFIX_ROWS=0
    GEA_EMBEDDED_DISPLAY_DIRECT_PSRAM_ALIGN_TRANSFERS=1
    # Real direct-PSRAM: packed stride (410 px, 820 B/row) keeps the framebuffer
    # contiguous so flushFramebufferRectDirect takes the directBytes branch and
    # GDMA reads PSRAM with no internal-RAM staging. CS_HELD_STREAM=1 above keeps
    # the SPI tx open across the whole frame so the touch task can't yield
    # between segments and starve the FIFO. REQUIRE_LENGTH_ALIGNMENT=0 lets the
    # final 6-row tail also flow direct from PSRAM (vs being staged through an
    # internal-RAM buffer). The GDMA hardware accepts non-64-aligned descriptor
    # tails; the matching esp_cache_msync uses ESP_CACHE_MSYNC_FLAG_UNALIGNED.
    GEA_EMBEDDED_DISPLAY_DIRECT_PSRAM_REQUIRE_LENGTH_ALIGNMENT=0
    GEA_EMBEDDED_DISPLAY_SOFTWARE_SCROLL_PADDED_STRIDE=0
    GEA_EMBEDDED_DISPLAY_DIRECT_PSRAM_PADDED_ROW_STREAM=0
    # Only relevant when DIRECT_PSRAM_FLUSH=1 (disabled above). Left at 0.
    GEA_EMBEDDED_CO5300_SPI_MANUAL_DMA_ALIGN_FLAG=0
    # Was =1 from an old display-pacing experiment (f06ad9724) that disabled the
    # then-broken queueFrameEvent catch-up. =0 re-enables catch-up, now driven by
    # the work-based, event-loop graceful-overrun path (frame_scheduler/runtime).
    GEA_EMBEDDED_FRAME_SCHEDULER_DROP_CATCHUP_FRAMES=0
    GEA_EMBEDDED_APP_FRAME_TASK_PRIORITY=23
    # (perf-lite and all other perf sub-flags are governed by GEA_EMBEDDED_PERF=0 above.)
    # 2-core replay split: defaults (96 rows / 64k px) only engage at the largest
    # dirty phases of an animated-transform scene; lower so mid-size regions split
    # too. Measured on css-3d-cube (glassy): median 52.3 -> 59.2 fps, min 44 -> 49.
    GEA_EMBEDDED_RENDER_PARALLEL_MIN_ROWS=48
    GEA_EMBEDDED_RENDER_PARALLEL_MIN_PIXELS=24000
    ${GEA_EMBEDDED_DIRECT_CANVAS_COMPILE_DEFINITIONS}
    ${GEA_EMBEDDED_BLE_COMPILE_DEFINITIONS}
    ${GEA_EMBEDDED_HTTPS_COMPILE_DEFINITIONS}
    ${GEA_EMBEDDED_CAPABILITY_COMPILE_DEFINITIONS}
    # Serve device logs over TCP 8081 (`scripts/board ota-monitor` already speaks
    # the framing). This board has the internal DRAM for the ~6 KiB task since
    # the IPC task stacks were cut from 16 KiB to 4 KiB, and without it a board
    # whose USB console has dropped has no way to show a log at all.
    GEA_EMBEDDED_DIAGNOSTICS_ENABLED=1
)

# --- App compilation: vite-plugin-gea -> geatsc ---

_gea_collect_app_sources(_ACTIVE_APP_SRCS "${GEA_EMBEDDED_APP_DIR}")
_gea_collect_app_assets(_ACTIVE_APP_ASSETS "${GEA_EMBEDDED_APP_DIR}")
# The Gea compiler emits JS and IR, then geatsc emits C++. We use a stamp file as
# the canonical OUTPUT because geatsc now writes a source manifest plus multiple
# translation units. The generated archive consumes geatsc-sources.txt after this
# command has completed.
set(_GEATSC_BIN "${GEA_COMPILER}/dist/cli.js")
set(_GEATSC_BUILD_STAMP "${GENERATED_APP_DIR}/geatsc_build.stamp")
set(_GEA_VITE_GEATSC_SCRIPT "${GEA_CORE}/scripts/build-gea-vite-geatsc.mjs")
file(GLOB_RECURSE _GEATSC_DIST_SRCS CONFIGURE_DEPENDS
    "${GEA_COMPILER}/dist/*.js"
    "${GEA_COMPILER}/dist/targets/cpp/runtime/*.cpp"
    "${GEA_PLUGIN}/dist/*.js"
)
add_custom_command(
    OUTPUT ${_GEATSC_BUILD_STAMP}
    COMMAND ${CMAKE_COMMAND} -E env
            "GEA_EMBEDDED_DIRECT_CANVAS_CONTEXT=${GEA_EMBEDDED_DIRECT_CANVAS_CODEGEN}"
            "GEA_EMBEDDED_NUMBER_F32=${GEA_EMBEDDED_NUMBER_F32}"
            node ${_GEA_VITE_GEATSC_SCRIPT}
            --app-dir ${GEA_EMBEDDED_APP_DIR}
            --entry ${GEA_EMBEDDED_APP_ENTRY}
            --cpp-board "${GEA_EMBEDDED_CPP_BOARD}"
            --out-dir ${GENERATED_APP_DIR}
            --geatsc-bin ${_GEATSC_BIN}
            --font-device-pixel-ratio ${GEA_EMBEDDED_CSS_DEVICE_PIXEL_RATIO}
            --pixel-panel-endian 1
            --gea-ir-backend
            # gea.compilerPlugins: an app's own geatsc plugins, e.g. the one that
            # declares its native host-function boundary. Empty for most apps.
            ${GEA_EMBEDDED_APP_GEATSC_PLUGIN_ARGS}
    COMMAND ${CMAKE_COMMAND} -E touch ${_GEATSC_BUILD_STAMP}
    DEPENDS
        ${GEA_EMBEDDED_APP_GEATSC_PLUGINS}
        ${_ACTIVE_APP_SRCS}
        ${_ACTIVE_APP_ASSETS}
        ${_GEATSC_BIN}
        ${_GEATSC_DIST_SRCS}
        ${_GEA_COMPILER_DEP_SRCS}
        ${_GEA_FONT_ASSETS}
    BYPRODUCTS
        "${GEATSC_SOURCE_LIST}"
        "${GENERATED_APP_DIR}/entry.cpp"
        "${GENERATED_APP_DIR}/generated_support.hpp"
        "${GENERATED_APP_DIR}/runtime.cpp"
        "${GENERATED_APP_DIR}/runtime_pch.h"
        ${GENERATED_CONFIG_H}
        ${GENERATED_FONT_CPP}
        ${GENERATED_FONT_H}
        ${GENERATED_ASSETS_CPP}
        "${GENERATED_APP_DIR}/gea-vite-bundle.js"
    COMMENT "Compiling ${GEA_EMBEDDED_APP}: TSX -> JS via vite-plugin-gea -> C++ via geatsc"
    VERBATIM
)
add_custom_target(geatsc_build ALL DEPENDS ${_GEATSC_BUILD_STAMP})
gea_geatsc_add_generated_archive(
    TARGET ${COMPONENT_LIB}
    NAME geatsc_active_archive
    SOURCE_LIST "${GEATSC_SOURCE_LIST}"
    OUTPUT "${GEATSC_ARCHIVE}"
    DEPENDS ${_GEATSC_BUILD_STAMP}
    COMPILE_OPTIONS ${GEATSC_COMPILE_OPTIONS}
    COMPILE_DEFINITIONS ${GEATSC_COMPILE_DEFINITIONS}
)
# The Unix Makefiles generator emits the geatsc_build.stamp rule in a
# separate sub-make from the archive target. On a cold build both reach the
# stamp independently and run the TSX->C++ compile twice in parallel, racing
# on the shared apps/<app>/module-graph dir (ENOTEMPTY). Order the archive
# after geatsc_build so the stamp is already fresh and the compile runs once.
add_dependencies(geatsc_active_archive geatsc_build)

# Embedded image assets (gea_asset_*): the generated TU defines weak const
# arrays referenced from libgea_active_geatsc.a. Weak-only archive members are
# never extracted to resolve a cross-archive reference, so force the dedicated
# asset archive in. Keep the archive and both linker-state switches in one
# argument: otherwise CMake can separate the switches across IDF's repeated
# dependency groups. --gc-sections still drops unused asset sections.
add_library(gea_active_app_assets STATIC "${GENERATED_ASSETS_CPP}")
add_dependencies(gea_active_app_assets geatsc_build)
target_link_libraries(${COMPONENT_LIB} PRIVATE
    "-Wl$<COMMA>--whole-archive$<COMMA>$<TARGET_FILE:gea_active_app_assets>$<COMMA>--no-whole-archive")
